\begin{frame}{Introduction}

\textbf{Research question.} Do firms with greater \textit{pre-mandate} human capital
disclosure (HCD) experience a larger increase in their earnings response coefficient
(ERC) after the SEC's 2020 mandate?

\vspace{4pt}

\begin{beamercolorbox}[sep=4pt,rounded=true]{block body}
{\footnotesize \textbf{ERC --- a quick gloss.}\\[2pt]
\textit{Conceptually:} how strongly the stock market revises its valuation
when a firm announces earnings. High ERC $\Rightarrow$ investors treat
earnings as an informative, credible signal of firm value; low ERC $\Rightarrow$
earnings move prices little (noisy, or already known).\\[2pt]
\textit{Operationally:} the slope $\beta$ in
$\text{Return} = \alpha + \beta \cdot \text{Earnings surprise} + \varepsilon$.}
\end{beamercolorbox}

\vspace{2pt}

\textbf{Background.}
\begin{itemize}
  \item Pre-2020: SEC required only employee \textit{headcount} in 10-Ks (checklist regime).
  \item Nov 2020 --- Reg S-K Item 101(c) shifts to a \textit{principles-based} regime:
        firms must describe ``human capital resources'' material to their business.
  \item No specific metrics prescribed $\Rightarrow$ wide cross-sectional variation
        in disclosure depth across firms.
\end{itemize}

\end{frame}

% SLIDE 4: Data
% =========================================================
\begin{frame}{Data}

\textbf{Sources}
\begin{itemize}
  \item \textbf{SEC EDGAR}: 10-K filings, FY2015--2024 (text)
  \item \textbf{Compustat}: firm financials (earnings, market value)
  \item \textbf{CRSP}: monthly returns (for buy-and-hold abnormal returns)
\end{itemize}

\textbf{Pipeline (Python)}
\begin{enumerate}
  \item Pull 10-K filings via the EDGAR REST API (rate-limited, locally cached).
  \item Extract Item~1 (``Business'') with regex + iXBRL cleaning.
  \item Tokenize; greedy phrase-match against the Demers, Wang \& Wu (2024) lexicon
        (1{,}285 HC keywords across 5 categories).
  \item Compute filing-level intensity:
        $hc\_intensity_{i,t} = \frac{\text{\#\,HC keywords in Item 1}}{\text{\#\,words in Item 1}}$.
  \item Merge with Compustat and CRSP at the firm-year level.
\end{enumerate}

\textbf{Final sample:} $\sim 33{,}000$ firm-year observations, $\sim 4{,}000$ unique firms.

\end{frame}

% =========================================================
% SLIDE 4: Methods
% =========================================================
\begin{frame}{Methods}

\textbf{Treatment exposure} (predetermined, FY2015--2019, standardized):
\[
HCD_i \;=\; \frac{1}{5}\sum_{t=2015}^{2019} hc\_intensity_{i,t}
\]

\textbf{Triple-interaction regression:}
{\small
\begin{align*}
AR_{i,t} =\; \alpha
&+ \beta_1\, UE_{i,t}
\;+\; \beta_2\, Post_t \times UE_{i,t}
\;+\; \beta_3\, HCD_i \times UE_{i,t}\\
&+ \beta_4\, Post_t \times HCD_i
\;+\; \boldsymbol{\beta_5\, Post_t \times HCD_i \times UE_{i,t}}\\
&+ \mu_i + \lambda_t + \varepsilon_{i,t}
\end{align*}}

\begin{itemize}
  \item $AR_{i,t}$: 9-month buy-and-hold market-adjusted return (3 months after FYE)
  \item $UE_{i,t}$: earnings ($IB$) scaled by lagged market value of equity
  \item $Post_t = \mathbb{1}[t \geq 2021]$
  \item Firm FE $\mu_i$ + year FE $\lambda_t$; SE clustered by firm; estimated via \texttt{linearmodels.PanelOLS}
\end{itemize}

\textbf{Prediction (H1):} $\hat{\beta}_5 > 0$ --- firms with deeper pre-mandate HCD experience a larger ERC increase after the mandate.

\end{frame}

% SLIDE 6: Takeaways
% =========================================================
\begin{frame}{What did we learn? Why does it matter?}

\textbf{Empirical finding}
\begin{itemize}
  \item Firms with deeper voluntary HCD \textit{before} the mandate experience a
        differentially larger ERC increase \textit{after} it.
\end{itemize}

\textbf{Why it matters}
\begin{itemize}
  \item \textbf{Disclosure regulation design:} principles-based mandates are
        \textit{not} neutral --- their value relevance depends on firms' prior
        disclosure infrastructure.
  \item \textbf{Capital markets:} voluntary disclosure becomes more informative
        once formalized through mandatory channels (legal liability, audit).
  \item \textbf{Policy:} informs ongoing debate around the SEC's 2022 proposal
        to expand mandatory human-capital metrics.
\end{itemize}

\textbf{Limitations \& next steps}
\begin{itemize}
  \item Add firm-level controls (size, leverage, analyst coverage, IO).
  \item Run an event-study / parallel-trends specification.
\end{itemize}
